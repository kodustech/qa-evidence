/* QA #2420 A2 E-h: real kodyRuleEvidenceVerifier + hasRuleEvidence for the
 * file-scoped PR-title rule E. Stubbed: the evidence-collection agent run
 * (returns keep=true, reads scripted), the judge LLM.run (verdict scripted). */
import * as fs from 'fs';
import { LLM } from '@libs/llm/llm';
import { kodyRuleEvidenceVerifier, hasRuleEvidence } from './kody-rule-evidence';

const reads: Array<{ path: string; content: string }> = [];
jest.mock('@libs/agent-harness/infrastructure/ai-sdk/ai-sdk-agent-runner', () => ({
    AiSdkAgentRunner: class {
        async run(_spec: any, _input: any, opts: any) {
            for (const r of reads) opts.services.readEvidence.push(r);
            return { status: 'completed' };
        }
    },
}));
jest.mock('@libs/code-review/infrastructure/agents/core/verifier.agent', () => {
    const actual = jest.requireActual('@libs/code-review/infrastructure/agents/core/verifier.agent');
    return { ...actual, extractVerdict: () => ({ keep: true, confidence: 'low', rationale: 'uncertain' }) };
});

const OUT = process.env.QA_OUT;
const records: any[] = [];
afterAll(() => OUT && fs.writeFileSync(OUT, JSON.stringify(records, null, 2)));

const RULE_E = {
    uuid: 'qa2420-E',
    title: 'QA2420 PR title prefix (file scope)',
    rule: 'The pull request title must start with `[qa2420-file]` and name the module and the operation it changes.',
    scope: 'file',
    contextNeed: null,
};
const CHECKIN = 'qa2420/CheckinReport.cs';
const input: any = {
    organizationAndTeamData: { organizationId: '11111111-1111-4111-8111-111111111111', teamId: '22222222-2222-4222-8222-222222222222' },
    prNumber: 2,
    prTitle: 'update',
    prBody: '',
    changedFiles: [{ filename: CHECKIN, patch: '@@ -0,0 +1,2 @@\n+using System;\n+namespace Ares.Qa2420;' }],
    prFilePaths: [CHECKIN],
    remoteCommands: {},
};
const candidate = {
    ruleId: '1',
    relevantLinesStart: null,
    relevantLinesEnd: null,
    suggestionContent: 'The PR title "update" does not start with [qa2420-file].',
    oneSentenceSummary: 'PR title lacks [qa2420-file]',
};
const judge = (path: string, line: number) => ({
    rationale: 'Title is "update".',
    regressionCoverageClaim: false,
    existingRegressionPath: '',
    existingRegressionLine: 0,
    verdict: 'violated',
    path,
    line,
});

describe('E-h evidence gate on the file-scoped title rule', () => {
    const cases: Array<[string, string | null, any, boolean, any[]]> = [
        ['file shard (rule scope file), collector read nothing, judge cites "PR title":0', CHECKIN, judge('PR title', 0), true, []],
        ['file shard (rule scope file), collector read the file, judge cites "PR title":0 (live R8/R10 shape)', CHECKIN, judge('PR title', 0), true, [{ path: CHECKIN, content: '1: using System;\n2: namespace Ares.Qa2420;' }]],
        ['PR shard (rule scope pull-request), judge cites "PR title":0', null, judge('PR title', 0), true, []],
        ['file shard, judge cites the file line it read (CheckinReport.cs:1)', CHECKIN, judge(CHECKIN, 1), true, [{ path: CHECKIN, content: '1: using System;\n2: namespace Ares.Qa2420;' }]],
        ['file shard, judge cites "PR title":0, sandbox unavailable', CHECKIN, judge('PR title', 0), false, []],
    ];
    it.each(cases)('%s', async (name, filename, verdict, repoAvailable, rd) => {
        reads.length = 0;
        reads.push(...rd);
        const spy = jest.spyOn(LLM, 'run').mockResolvedValue(verdict as any);
        const verify = kodyRuleEvidenceVerifier(input, { provider: 'openai', model: 'scripted' } as any, repoAvailable);
        const shard: any = { filename, ruleUnits: [RULE_E], ruleUuids: ['qa2420-E'] };
        const reason = await verify(candidate as any, shard, Date.now() + 60_000);
        const judgeInput = spy.mock.calls[0] ? JSON.parse((spy.mock.calls[0][0] as any).user) : null;
        records.push({
            case: name,
            shardFilename: filename,
            metadataAllowed: filename === null && !(candidate as any).needsRepositoryEvidence,
            judgeVerdict: verdict,
            reason,
            judgeCalls: spy.mock.calls.length,
            judgeSawTitle: judgeInput?.title,
            judgeLocation: judgeInput?.location,
            hasRuleEvidenceDirect: hasRuleEvidence(verdict, filename === null, rd),
        });
        spy.mockRestore();
    });
});
