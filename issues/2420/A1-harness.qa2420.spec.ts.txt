const route = (method: string, path: string, body: any): [number, any] => {
    if (path.startsWith('/bb/')) {
        if (method === 'POST' && /\/pullrequests\/\d+\/comments$/.test(path)) {
            const id = ++idSeq;
            return [201, { id, content: { raw: body?.content?.raw }, inline: body?.inline, parent: body?.parent }];
        }
        if (/\/pullrequests\/\d+\/commits$/.test(path))
            return [200, { values: [{ hash: 'headsha', date: now, message: 'm', author: { raw: 'qa <qa@example.invalid>' }, parents: [] }] }];
        return [200, { values: [] }];
    }
    if (path.startsWith('/az/')) {
        if (method === 'POST' && /\/threads$/.test(path)) {
            const threadId = ++azThreadSeq;
            // Azure numbers comments inside each thread from 1.
            return [200, { id: threadId, comments: [{ id: 1, content: body?.comments?.[0]?.content }], threadContext: body?.threadContext ?? null, status: 'active' }];
        }
        if (/\/commits$/.test(path))
            return [200, { value: [{ commitId: 'headsha', comment: 'm', author: { name: 'qa', email: 'qa@example.invalid', date: now }, committer: { name: 'qa', date: now } }] }];
        if (/\/iterations$/.test(path)) return [200, { value: [{ id: 1, sourceRefCommit: { commitId: 'headsha' }, targetRefCommit: { commitId: 'basesha' } }] }];
        return [200, { value: [] }];
    }
    if (path.startsWith('/fj/')) {
        if (method === 'POST' && /\/issues\/\d+\/comments$/.test(path)) {
            const id = ++idSeq;
            return [201, { id, body: body?.body, html_url: `http://fake/fj/c/${id}` }];
        }
        if (method === 'POST' && /\/pulls\/\d+\/reviews$/.test(path)) {
            const id = ++idSeq;
            return [200, { id, body: body?.body, comments: body?.comments }];
        }
        if (/\/pulls\/\d+\/commits$/.test(path))
            return [200, [{ sha: 'headsha', commit: { message: 'm', author: { name: 'qa', email: 'qa@example.invalid', date: now }, committer: { date: now } }, parents: [] }]];
        if (/\/pulls\/\d+\/reviews\/\d+\/comments$/.test(path)) return [200, [{ id: idSeq + 1, path: 'x', position: 1 }]];
        return [200, []];
    }
    return [404, {}];
};

    requests = [];
    let ctx = await (stage as any).executeStage(baseCtx(providerKey, changedFiles));
    out.round1 = {
        agentStage: {
            validSuggestionsByPR: (ctx.validSuggestionsByPR ?? []).map((s: any) => ({ ...pickS(s), agentPrompt: resolveAgentPrompt(s) })),
            validSuggestions: (ctx.validSuggestions ?? []).map(pickS),
            discarded: (ctx.discardedSuggestions ?? []).map(pickS),
            logs: logs.filter((l) => /KODY-DEDUP|Snapped|File-anchored|Rendered|REPEAT/.test(l.message ?? '')),
        },
    };
    // formatter output on its own (same call the comment manager makes)
    out.round1.formatReviewCommentBody = [];
    for (const s of ctx.validSuggestionsByPR ?? []) {
        out.round1.formatReviewCommentBody.push({
            rules: s.brokenKodyRulesIds,
            body: await w.codeManagement.formatReviewCommentBody(
                { suggestion: s, repository: { name: p.repoName, language: 'csharp' }, includeHeader: true, includeFooter: false, language: 'en-US', organizationAndTeamData: ORG, suggestionCopyPrompt: true },
                p.type,
            ),
        });
    }
    const prLevelStage = new CreatePrLevelCommentsStage(w.commentManager as any, w.suggestionService as any, w.pullRequestsService as any);
    const fileStage = new CreateFileCommentsStage(w.commentManager as any, w.pullRequestsService as any, w.suggestionService as any);
    const lineSpy = jest.spyOn(w.commentManager, 'createLineComments');
    ctx = await (prLevelStage as any).executeStage(ctx);
    const afterPrLevel = requests.length;
    ctx = await (fileStage as any).executeStage(ctx);
    const lineResults = await (lineSpy.mock.results[0]?.value ?? Promise.resolve(null));
    out.round1.requests = requests.map((r, i) => ({ ...r, stage: i < afterPrLevel ? 'CreatePrLevelCommentsStage' : 'CreateFileCommentsStage', text: bodyOf(r) }));
    out.round1.bodyChecks = requests
        .filter((r) => /qa2420\/CheckinReport\.cs|CheckinReport/.test(bodyOf(r) ?? '') && !(r.provider === 'bb' && r.body?.parent))
        .map((r) => ({ path: r.path, general: r.provider === 'bb' ? !r.body?.inline : r.provider === 'az' ? !r.body?.threadContext : /\/issues\/\d+\/comments$/.test(r.path), inlineFields: r.provider === 'bb' ? r.body?.inline : r.provider === 'az' ? r.body?.threadContext : r.body?.comments?.map((c: any) => ({ path: c.path, new_position: c.new_position })), ...checkBody(bodyOf(r), p.slash) }));
    out.round1.promptReplies = requests.filter((r) => r.provider === 'bb' && r.body?.parent).map((r) => ({ parent: r.body.parent, text: r.body?.content?.raw, ...checkBody(r.body?.content?.raw, p.slash) }));
    out.round1.storedPrLevel = r1.doc.prLevelSuggestions;
    out.round1.lineCommentResults = lineResults?.commentResults?.map((c: any) => ({ deliveryStatus: c.deliveryStatus, commentId: c.codeReviewFeedbackData?.commentId, path: c.comment?.path, line: c.comment?.line, start_line: c.comment?.start_line, suggestionId: c.comment?.suggestion?.id })) ?? null;
    out.round1.ctxErrors = ctx.errors;
    out.round1.repoCalls = [...new Set(r1.calls)];

    // the inline D (P) becomes a file-scoped sent record, as the file stage's persistence would store it
    for (const c of lineResults?.commentResults ?? []) {
        if (c.deliveryStatus !== DeliveryStatus.SENT) continue;
        const s = c.comment.suggestion;
        // The persisted file suggestion carries a unique id (Mongo doc); the stage output has none here, so a synthetic one stands in.
        r1.fileSent.push({ ...s, id: s.id ?? `file-suggestion-${c.codeReviewFeedbackData?.commentId}`, deliveryStatus: DeliveryStatus.SENT, createdAt: new Date(Date.now()).toISOString(), comment: { id: c.codeReviewFeedbackData?.commentId } });
    }

    // round 2: same findings re-emitted, history read by the real use case + store
    out.round2 = {};
    const store = new PrDecisionStoreService(r1.repo);
    const useCase = new BuildPreviousReviewDecisionsUseCase(store);
    const history = await useCase.execute({ organizationId: ORG.organizationId, prNumber: 2420, repositoryFullName: 'qa-owner/qa-repo' } as any);
