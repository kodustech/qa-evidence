const makeStage = (orchestratorCalls: any[]) => {
    const reviewOrchestrator = {
        execute: jest.fn(async (input: any) => {
            orchestratorCalls.push({ agentScope: input.agentScope, forcedModel: input.forcedModel?.model, kodyRules: (input.kodyRules ?? []).length });
            const base = { agentResults: [], failures: [], incomplete: [], unverified: [], warnings: [] };
            if (input.agentScope === 'auxiliary') return { ...base, suggestions: [{ ...ruleT }, { ...ruleB }] };
            if (input.agentScope === 'native') return { ...base, suggestions: [bug(input.forcedModel?.model ?? '?')] };
            return { ...base, suggestions: [{ ...ruleT }, { ...ruleB }, bug('balanced')] };
        }),
    };
    const stage = new AgentReviewStage(
        { findLatestStageLog: jest.fn(), updateCodeReview: jest.fn().mockResolvedValue(null), updateStageLog: jest.fn() } as any,
        { findByExternalId: jest.fn().mockResolvedValue(null) } as any,
        reviewOrchestrator as any,
        { runLLMInSpan: jest.fn(async ({ runFn }: any) => runFn?.()) } as any,
        { generateContext: jest.fn(), generateContextLegacy: jest.fn().mockResolvedValue('') } as any,
        { isEnabled: jest.fn().mockResolvedValue(false) } as any,
        { getReleaseTrack: jest.fn().mockResolvedValue('stable') } as any,
        { getRepositories: jest.fn().mockResolvedValue([]), getCloneParams: jest.fn().mockResolvedValue(null), getRepositoryContentFile: jest.fn().mockResolvedValue(null), getRepositoryAllFiles: jest.fn().mockResolvedValue([]) } as any,
    );
    const logs: string[] = [];
    const sink = (level: string) => (e: any) => logs.push(`${level}: ${e?.message}`);
    (stage as any).logger = { log: sink('log'), warn: sink('warn'), error: sink('error'), debug: () => {}, verbose: () => {} };
    return { stage, logs };
};
