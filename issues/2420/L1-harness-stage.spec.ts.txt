const pick = (s: any) =>
    s && {
        file: s.relevantFile,
        start: s.relevantLinesStart,
        end: s.relevantLinesEnd,
        label: s.label,
        rule: s.brokenKodyRulesIds?.[0],
        fileAnchored: s.fileAnchored,
        content: s.suggestionContent,
        fullExplanation: s.fullExplanation,
        kodyRuleOtherLocations: s.kodyRuleOtherLocations,
        priorityStatus: s.priorityStatus,
    };

const summarize = (result: any, logs: string[], llmCalls: number) => ({
    byPR: (result.validSuggestionsByPR ?? []).map((s: any) => ({
        content: s.suggestionContent,
        fullExplanation: s.fullExplanation,
        prompt: resolveAgentPrompt(s),
        label: s.label,
        rule: s.brokenKodyRulesIds?.[0],
    })),
    valid: (result.validSuggestions ?? []).map(pick),
    inline: (result.fileAnalysisResults ?? []).flatMap((f: any) =>
        (f.validSuggestionsToAnalyze ?? []).map((s: any) => ({
            ...pick(s),
            commentStartLine: calculateCommentStartLine(s),
            commentLine: calculateCommentEndLine(s),
        })),
    ),
    discarded: (result.discardedSuggestions ?? []).map((s: any) => ({
        file: s.relevantFile,
        start: s.relevantLinesStart,
        end: s.relevantLinesEnd,
        label: s.label,
        rule: s.brokenKodyRulesIds?.[0],
        status: s.priorityStatus,
    })),
    logs: logs.filter((l) => /Snapped|File-anchored|Dropped|KODY-DEDUP|Rendered merged|PARSE-SCHEMA|PATH_MISMATCH|Completed partial|not changed/.test(l)),
    modelCalls: llmCalls,
});

let llmSpy: jest.SpyInstance;

const modelCallCount = () =>
    llmSpy.mock.calls.length +
    (classifySeverity as jest.Mock).mock.calls.length +
    (formatSuggestionContent as jest.Mock).mock.calls.length;

const runStage = async (
    scenario: string,
    name: string,
    suggestions: any[],
    ctxOver: Record<string, unknown> = {},
    opts: { gate?: boolean; cfgOver?: Record<string, unknown>; extraOut?: any } = {},
) => {
    const { stage, reviewOrchestrator, logs } = makeStage(opts.gate);
    const anchorSpy = jest.spyOn(stage as any, 'anchorSuggestionsToDiff');
    reviewOrchestrator.execute.mockResolvedValue({
        suggestions,
        agentResults: [],
        failures: [],
        incomplete: [],
        unverified: [],
        warnings: [],
        ...(opts.extraOut ?? {}),
    });
    const ctx = makeContext(ctxOver, opts.cfgOver);
    const result = await (stage as any).executeStage(ctx);
    const anchorCall = anchorSpy.mock.calls[0];
    const out = {
        scenario,
        case: name,
        input: suggestions.map(pick),
        anchorArgCount: anchorCall?.length,
        anchorSetArg: anchorCall?.[5] instanceof Set ? [...anchorCall[5]] : String(anchorCall?.[5]),
        anchorOut: (anchorSpy.mock.results[0]?.value ?? []).map(pick),
        ...summarize(result, logs, modelCallCount()),
    };
    return rec(out);
};

// ── M29 local: zero model calls in anchor/merge, bounded 1,000 merge ───────
describe('M29 cost', () => {
    it('anchor helper and merge make zero model calls', () => {
        const { stage } = makeStage();
        const before = modelCallCount();
        (stage as any).anchorSuggestionsToDiff(
            [finding(), finding({ relevantLinesStart: 0 }), finding({ relevantLinesStart: 34, relevantLinesEnd: 34 })],
            [{ filename: F1, patch: TWO_HUNKS }], RULES, [], {}, new Set([F1]),
        );
        mergeKodyRuleFindings(GROUPS['G1-kept-lineless-plus-lined-dups'].map((x) => ({ ...x })));
        rec({ scenario: 'M29', case: 'zero-model-calls', modelCallsDuringAnchorAndMerge: modelCallCount() - before });
    });
    it('1,000-occurrence merge', () => {
        const files = Array.from({ length: 50 }, (_, i) => `qa2420/f${i}.ts`);
        const input: any[] = [];
        for (let i = 0; i < 1000; i++) {
            const rule = `rule-${i % 10}`;
            const file = files[i % 50];
            const kind = i % 4; // 0 lined, 1 lineless-file, 2 no-file, 3 start-only
            input.push(mf(kind === 2 ? undefined : file, kind === 1 || kind === 2 ? undefined : (i % 97) + 1, kind === 0 ? (i % 97) + 2 : undefined, `finding ${i} ${'x'.repeat(i % 13)}`, rule));
        }
        const merges: any[] = [];
        const t0 = process.hrtime.bigint();
        const before = modelCallCount();
        const out = mergeKodyRuleFindings(input, (m) => merges.push(m));
        const ms = Number(process.hrtime.bigint() - t0) / 1e6;
        const validLocs = new Set(input.filter((s) => s.relevantFile && s.relevantLinesStart).map((s) => `${s.relevantFile}:${s.relevantLinesStart}-${s.relevantLinesEnd ?? s.relevantLinesStart}`));
        const allOther = out.flatMap((s: any) => s.kodyRuleOtherLocations ?? []);
        const invented = allOther.filter((l: string) => !validLocs.has(l));
        const rendered = out.flatMap((s: any) => otherLocationsToRender(s));
        rec({
            scenario: 'M29', case: 'merge-1000', ms, outCount: out.length, merges: merges.length,
            scopes: merges.reduce((a: any, m) => ((a[m.scope] = (a[m.scope] || 0) + 1), a), {}),
            otherLocationsTotal: allOther.length,
            invented: invented.length, inventedSample: invented.slice(0, 5),
            placeholderRendered: rendered.filter((l: string) => /\?|undefined|null/.test(l)).length,
            modelCalls: modelCallCount() - before,
        });
    });
});
