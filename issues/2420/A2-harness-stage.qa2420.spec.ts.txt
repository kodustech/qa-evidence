type Script = 'same' | 'new_place' | 'different' | 'undecided' | 'fail';
let script: Script = 'same';

const answer = (user: string) => {
    const blocks = user.split('<Finding n="').slice(1);
    const verdicts = blocks.map((b) => {
        const n = Number(b.split('"')[0]);
        const earlier = /<Earlier id="([^"]+)"/.exec(b)?.[1] ?? null;
        if (script === 'same') return { finding: n, verdict: 'same_occurrence', earlierId: earlier };
        if (script === 'new_place') return { finding: n, verdict: 'new_place', earlierId: earlier };
        return { finding: n, verdict: 'different_problem', earlierId: null };
    });
    return { verdicts };
};

const runStage = async (suggestions: any[], opts: { ctxOver?: any; cfgOver?: any; buildPrev?: any } = {}) => {
    const { stage, reviewOrchestrator, logs } = makeStage(opts.buildPrev);
    const anchorSpy = jest.spyOn(stage as any, 'anchorSuggestionsToDiff');
    reviewOrchestrator.execute.mockResolvedValue({ suggestions, agentResults: [], failures: [], incomplete: [], unverified: [], warnings: [] });
    const result = await (stage as any).executeStage(makeContext(opts.ctxOver, opts.cfgOver));
    return {
        result,
        anchored: (anchorSpy.mock.results[0]?.value ?? []).map((s: any) => ({ ...s })),
        out: {
            byPR: (result.validSuggestionsByPR ?? []).map((s: any) => ({ ...s })),
            inline: (result.validSuggestions ?? []).map(pick),
            discarded: (result.discardedSuggestions ?? []).map(pick),
            logs: logs.filter((l) => /Snapped|File-anchored|Dropped|KODY-DEDUP|REPEAT-CHECK|Rendered merged/.test(l)),
            repeatCheckCalls: repeatPrompts.length,
            runNames: [...runNames],
        },
    };
};

// ── persistence (in-memory PR doc, shapes per pullRequests.model.ts) ───────
const persist = (result: any) => {
    const files: Record<string, any[]> = {};
    let i = 0;
    for (const s of result.validSuggestions ?? []) {
        (files[s.relevantFile] ??= []).push({
            ...s,
            id: s.id ?? `file-sugg-${++i}`,
            deliveryStatus: DeliveryStatus.SENT,
            comment: { id: 900000 + i, pullRequestReviewId: null },
            createdAt: '2026-11-25T03:44:00.000Z',
        });
    }
    // PR-level: only the fields the prLevelSuggestions schema carries.
    const prLevelSuggestions = (result.validSuggestionsByPR ?? []).map((s: any, k: number) => ({
        id: s.id,
        suggestionContent: s.suggestionContent,
        fullExplanation: s.fullExplanation,
        oneSentenceSummary: s.oneSentenceSummary,
        label: s.label,
        severity: s.severity,
        brokenKodyRulesIds: s.brokenKodyRulesIds,
        deliveryStatus: DeliveryStatus.SENT,
        comment: { id: 4317603269 + k, pullRequestReviewId: null },
        createdAt: '2026-11-25T03:44:00.000Z',
    }));
    return { files: Object.entries(files).map(([filename, suggestions]) => ({ filename, path: filename, suggestions })), prLevelSuggestions };
};
const repoFor = (doc: any) => ({
    findSuggestionsOnPR: jest.fn(async () => doc.files.flatMap((f: any) => f.suggestions).filter((s: any) => s.deliveryStatus === DeliveryStatus.SENT)),
    findPrLevelSuggestionsByPR: jest.fn(async () => doc.prLevelSuggestions.filter((s: any) => s.deliveryStatus === DeliveryStatus.SENT)),
});

    const history = await buildPrev.execute({ organizationId: ORG.organizationId, prNumber: 2, repositoryFullName: 'qa/ares' } as any);
    return { r1, doc, history, buildPrev };
};

const plausibleView = (f: any, e: any) => {
    const norm = (p?: string) => (p || '').trim().replace(/^\.?\//, '');
    return {
        earlierId: e.suggestionId,
        findingFile: norm(f.relevantFile),
        earlierFile: norm(e.relevantFile),
        fileMatch: norm(f.relevantFile) === norm(e.relevantFile),
        linesOverlap:
            f.relevantLinesStart != null && e.relevantLinesStart != null &&
            f.relevantLinesStart <= (e.relevantLinesEnd ?? e.relevantLinesStart) &&
            e.relevantLinesStart <= (f.relevantLinesEnd ?? f.relevantLinesStart),
        exactClaim: (f.suggestionContent ?? '').trim() === (e.suggestionContent ?? '').trim(),
        earlierClaimHead: (e.suggestionContent ?? '').slice(0, 60),
        findingKeys: occurrenceKeys(f),
        earlierKeys: occurrenceKeys(e),
    };
};
