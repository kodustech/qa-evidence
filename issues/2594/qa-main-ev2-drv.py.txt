def read(ledger):
    return [json.loads(l) for l in open(ledger)] if os.path.exists(ledger) else []
def waitfile(f, timeout=120):
    t=time.time()
    while not os.path.exists(f):
        if time.time()-t>timeout: raise TimeoutError(f)
        time.sleep(0.2)
